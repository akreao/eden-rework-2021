//===== rAthena Script =======================================
//= Eden Group: 2021 mission boards
//===== Description: =========================================
//= kRO's April 2021 Eden Group (Paradise Group) rework replaced the
//= old level-bracket missions with six mission boards, a champion hunt
//= board, a logistics officer and cave shortcut NPCs. This file adds:
//=   Mission boards        34-44,38 and 25-35,98  six boards, twice
//=     1~39, 40~69 and 70~99    town daily hunts, EXP only
//=     100~110 and 120~140      town daily hunts, EXP and Eden Coins
//=     Advanced (Lv130-190)     quests 19000-19064, EXP and Eden Coins
//=   Logistics Officer     29,35 and 37,96   30-item deliveries, hourly
//=   Rigel, Aker, Ayla, Minmin               cave shortcuts for members
//=                                           on a matching board hunt
//=   Will-o'-Wisps (Payon Cave 4F) and Old Crates (Shipwreck 1F)
//=                                           for missions 21019 and 21130
//=   Champion Hunt Board   16,33 and 19,96   champion hunts, daily
//=   Diligent Sogil        18,35 and 21,96   takes the champion reports
//=
//= Each board quest uses kRO's own quest id, so it shares its
//= cooldown with the town daily of the same id. Cooldowns reset at
//= 04:00; deliveries can be repeated after an hour.
//=
//= Part of the Ragnarok Offline mod "eden-rework-2021": it loads only
//= while the mod is on. It is all or nothing: while it is loaded,
//= the old mission boards, the old 100-140 mission NPCs and their
//= field NPCs are hidden, and these take their place.
//=
//= Sources: quest ids, cooldown ids, titles and kill counts are from
//= the kRO client (Dec 2025). Lv1-99 EXP is from twRO (ro.ntome.com
//= and Bahamut 2838486, which agree), Lv100-190 EXP and coins from
//= the Eden rework spec (Inven 3641 and 5207, twRO and thRO for the
//= coins), deliveries from Bahamut 2838486 and hazyforest. Anything
//= not confirmed by two sources is marked [unverified] where it is
//= used. The dialogue is written for this script, not translated
//= from kRO.
//===== Changelogs: ==========================================
//= 1.0 First version.
//============================================================

// Mission board. Each duplicate's hidden name ends in its board
// number: "#edmb<board><copy>", e.g. "#edmb1a". Board 7 is the champion
// hunt board; its hunts are reported to Diligent Sogil ("#edmb7r<copy>").
-	script	Eden Mission Board#edmb	-1,{
	.@b = atoi(charat(strnpcinfo(2), 4));
	.@name$ = "[" + .board$[.@b] + "]";
	if (charat(strnpcinfo(2), 5) == "r") {
		.@name$ = "[Diligent Sogil]";
		mes .@name$;
		if (!callfunc("F_HasEdenGroupMark")) {
			mes "Champion hunts are for Eden Group members. Talk to Secretary Lime Evenor if you'd like to join.";
			close;
		}
		for (.@i = .first[.@b]; .@i < .last[.@b]; ++.@i) {
			if (callsub(S_State, .@i) == 4)
				++.@ready;
		}
		if (!.@ready) {
			mes "I take the reports for the champion hunts on the board next to me.";
			mes "Bring down the champion you took on, then come and see me.";
			close;
		}
		mes "Back from a champion hunt? Let's see.";
		next;
		callsub L_ReportAll, .@b, .@name$;
		mes .@name$;
		mes "That's all I have for you. Each hunt opens again after 4:00.";
		close;
	}
	if (!callfunc("F_HasEdenGroupMark")) {
		mes .@name$;
		mes "- You need an ^4d4dff'Eden Group Mark'^000000 -";
		mes "- to take these missions. -";
		close;
	}
	while (1) {
		.@ready = 0;
		for (.@i = .first[.@b]; .@i < .last[.@b]; ++.@i) {
			if (callsub(S_State, .@i) == 4)
				++.@ready;
		}
		mes .@name$;
		mes "- Missions for Eden Group members. -";
		if (.@b == 6)
			mes "- Base level 130 and up. -";
		if (.@b == 7) {
			mes "- Hunt a champion, then report -";
			mes "- to Diligent Sogil. Each hunt -";
			mes "- can be done once a day. -";
		}
		mes "- Missions reset daily at 4:00. -";
		if (.@ready)
			mes "- ^00AA00" + .@ready + "^000000 finished, ready to report. -";
		next;
		switch (select("Browse missions", (.@ready && .@b != 7 ? "Report all finished missions" : ""), "Close")) {
		case 1:
			callsub L_Browse, .@b;
			break;
		case 2:
			callsub L_ReportAll, .@b, .@name$;
			break;
		case 3:
			close;
		}
	}
	end;

// Lists the board's areas, then the missions in one area.
L_Browse:
	.@b = getarg(0);
	.@name$ = "[" + .board$[.@b] + "]";
	for (.@i = .first[.@b]; .@i < .last[.@b]; ++.@i) {
		if (inarray(.@area$, .q_area$[.@i]) < 0)
			.@area$[getarraysize(.@area$)] = .q_area$[.@i];
	}
	while (1) {
		mes .@name$;
		mes "- Choose an area. -";
		next;
		.@a = select(implode(.@area$, ":") + ":Back") - 1;
		if (.@a == getarraysize(.@area$))
			return;
		while (1) {
			deletearray .@row;
			.@menu$ = "";
			for (.@i = .first[.@b]; .@i < .last[.@b]; ++.@i) {
				if (.q_area$[.@i] != .@area$[.@a])
					continue;
				.@row[getarraysize(.@row)] = .@i;
				.@s = callsub(S_State, .@i);
				.@t$ = (.q_menu$[.@i] != "" ? .q_menu$[.@i] : .q_title$[.@i]);
				.@t$ = replacestr(.@t$, ":", " -");	// a ":" would split the menu entry
				if (.@s == 1)
					.@menu$ += "^999999[Lv" + .q_lv[.@i] + "] " + .@t$ + "^000000:";
				else if (.@s == 2)
					.@menu$ += "^999999[Done] " + .@t$ + "^000000:";
				else if (.@s == 3)
					.@menu$ += "^0055FF[Active]^000000 " + .@t$ + ":";
				else if (.@s == 4)
					.@menu$ += "^00AA00[Report]^000000 " + .@t$ + ":";
				else
					.@menu$ += "[Lv" + .q_lv[.@i] + "] " + .@t$ + ":";
			}
			mes .@name$;
			mes "- " + .@area$[.@a] + " -";
			next;
			.@m = select(.@menu$ + "Back") - 1;
			if (.@m == getarraysize(.@row))
				break;
			callsub L_Mission, .@b, .@row[.@m];
		}
	}
	end;

// Shows one mission and what can be done with it now.
L_Mission:
	.@b = getarg(0);
	.@i = getarg(1);
	.@name$ = "[" + .board$[.@b] + "]";
	.@s = callsub(S_State, .@i);
	mes .@name$;
	mes "^0000FF" + .q_title$[.@i] + "^000000";
	if (.@b == 6)	// board 6's areas are level brackets, and the title is the map
		mes "Base level " + .q_lv[.@i] + "+";
	else
		mes "Area: " + .q_area$[.@i] + ", base level " + .q_lv[.@i] + "+";
	if (.q_goal$[.@i] != "")
		mes .q_goal$[.@i] + ".";
	if (.q_items$[.@i] != "") {
		explode(.@p$, .q_items$[.@i], ",");
		for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2)
			mes "Bring " + .@p$[.@k + 1] + " " + getitemname(atoi(.@p$[.@k])) + ".";
	}
	.@r$ = "Base EXP " + callfunc("F_InsertComma", .q_exp[.@i]);
	if (.q_jexp[.@i])
		.@r$ += ", Job EXP " + callfunc("F_InsertComma", .q_jexp[.@i]);
	if (.q_coin[.@i])
		.@r$ += ", " + callsub(S_Coins, .@i) + " " + getitemname(25223);
	.@rw$ = callsub(S_Rewards, .@i);
	if (.@rw$ != "") {
		explode(.@p$, .@rw$, ",");
		for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2)
			.@r$ += ", " + .@p$[.@k + 1] + " " + getitemname(atoi(.@p$[.@k]));
	}
	mes "Reward: " + .@r$ + ".";
	next;
	switch (.@s) {
	case 0:
		if (select("Accept this mission:Back") == 2)
			return;
		setquest .q_id[.@i];
		if (.q_id2[.@i] && checkquest(.q_id2[.@i], HAVEQUEST) < 0)
			setquest .q_id2[.@i];
		if (.q_need$[.@i] != "")
			setd .q_need$[.@i], 0;
		// Supplies the mission needs, topped up to the listed count.
		if (.q_give$[.@i] != "") {
			explode(.@p$, .q_give$[.@i], ",");
			for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2) {
				.@c = atoi(.@p$[.@k + 1]) - countitem(atoi(.@p$[.@k]));
				if (.@c > 0)
					getitem atoi(.@p$[.@k]), .@c;
			}
		}
		mes .@name$;
		mes "- Mission accepted. -";
		if (.@b == 7)
			mes "- Report to Diligent Sogil, next to this board. -";
		else
			mes "- Report back to a " + .board$[.@b] + ". -";
		next;
		return;
	case 1:
		mes .@name$;
		mes "- You need base level " + .q_lv[.@i] + " -";
		mes "- for this mission. -";
		next;
		return;
	case 2:
		mes .@name$;
		mes "- You have done this mission -";
		mes "- today. It opens again -";
		mes "- after 4:00. -";
		next;
		return;
	case 3:
		if (select("Keep going:Give up this mission") == 1)
			return;
		callsub S_Drop, .@i;
		mes .@name$;
		mes "- Mission dropped. You can -";
		mes "- take it again any time. -";
		next;
		return;
	case 4:
		if (.@b == 7) {
			mes .@name$;
			mes "- Hunt complete. Report to -";
			mes "- Diligent Sogil, next to this board. -";
			next;
			return;
		}
		if (select("Report this mission:Back") == 2)
			return;
		callsub L_Result, .@name$, .@i, callsub(S_Report, .@i);
		return;
	}
	return;

// Reports every finished mission on the board in one go.
L_ReportAll:
	.@b = getarg(0);
	for (.@i = .first[.@b]; .@i < .last[.@b]; ++.@i) {
		if (callsub(S_State, .@i) == 4)
			callsub L_Result, getarg(1), .@i, callsub(S_Report, .@i);
	}
	return;

// Tells the player how a report went: speaker, mission, S_Report's result.
L_Result:
	.@i = getarg(1);
	mes getarg(0);
	mes "^0000FF" + .q_title$[.@i] + "^000000";
	if (getarg(2) == 1)
		mes "- You don't have the items for this mission yet. -";
	else if (getarg(2) == 2)
		mes "- You're carrying too much to take the reward. Free up some weight and try again. -";
	else
		mes "- Mission reported. Thank you! -";
	next;
	return;

// 0 = can accept, 1 = level too low, 2 = cooling down, 3 = in progress,
// 4 = finished. Clears a cooldown that has run out.
S_State:
	.@i = getarg(0);
	.@h = checkquest(.q_id[.@i], HUNTING);
	.@h2 = (.q_id2[.@i] ? checkquest(.q_id2[.@i], HUNTING) : 2);
	if (.@h == 2 && .@h2 == 2) {
		// Finished only once the items and any other goal are there too.
		if (.q_need$[.@i] != "" && getd(.q_need$[.@i]) < .q_needn[.@i])
			return 3;
		if (.q_items$[.@i] != "") {
			explode(.@p$, .q_items$[.@i], ",");
			for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2) {
				if (countitem(atoi(.@p$[.@k])) < atoi(.@p$[.@k + 1]))
					return 3;
			}
		}
		return 4;
	}
	if (.@h >= 0 || (.q_id2[.@i] && .@h2 >= 0))
		return 3;
	.@t = checkquest(.q_cd[.@i], PLAYTIME);
	if (.@t == 2) {
		erasequest .q_cd[.@i];
		.@t = -1;
	}
	if (.@t >= 0)
		return 2;
	if (BaseLevel < .q_lv[.@i])
		return 1;
	return 0;

// A mission's Eden Coin reward, times the coin_multiplier setting.
S_Coins:
	return .q_coin[getarg(0)] * callfunc("F_ModSetting", "eden-rework-2021", "coin_multiplier", 1);

// A mission's item rewards ("item,count,..."). With the champion_tickets
// setting on, the Lv130 and Lv140 champion hunts also give a +7 Weapon
// Refine Ticket and a +7 Armor Refine Ticket. Not kRO: the mod adds them so
// Zerenia's refine ticket trades in the Paramarket can be reached without
// the cash shop.
S_Rewards:
	.@i = getarg(0);
	.@r$ = .q_rewards$[.@i];
	if (.q_board[.@i] == 7 && .q_lv[.@i] >= 130
	&& callfunc("F_ModSetting", "eden-rework-2021", "champion_tickets", 0))
		.@r$ += (.@r$ != "" ? "," : "") + "6230,1,6234,1";
	return .@r$;

// Pays a finished mission. 0 = done, 1 = items missing, 2 = overweight.
S_Report:
	.@i = getarg(0);
	if (.q_items$[.@i] != "") {
		explode(.@p$, .q_items$[.@i], ",");
		for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2) {
			if (countitem(atoi(.@p$[.@k])) < atoi(.@p$[.@k + 1]))
				return 1;
		}
	}
	.@n = 0;
	if (.q_coin[.@i]) {
		.@rid[0] = 25223;
		.@rcnt[0] = callsub(S_Coins, .@i);
		.@n = 1;
	}
	.@rw$ = callsub(S_Rewards, .@i);
	if (.@rw$ != "") {
		explode(.@r$, .@rw$, ",");
		for (.@k = 0; .@k < getarraysize(.@r$); .@k += 2) {
			.@rid[.@n] = atoi(.@r$[.@k]);
			.@rcnt[.@n] = atoi(.@r$[.@k + 1]);
			++.@n;
		}
	}
	if (.@n && !checkweight2(.@rid, .@rcnt))
		return 2;
	for (.@k = 0; .@k < getarraysize(.@p$); .@k += 2)
		delitem atoi(.@p$[.@k]), atoi(.@p$[.@k + 1]);
	callsub S_Drop, .@i;
	if (checkquest(.q_cd[.@i], HAVEQUEST) >= 0)
		erasequest .q_cd[.@i];
	setquest .q_cd[.@i];
	if (.q_jexp[.@i])
		getexp .q_exp[.@i], .q_jexp[.@i];
	else
		getexp .q_exp[.@i], 0;
	for (.@k = 0; .@k < .@n; ++.@k)
		getitem .@rid[.@k], .@rcnt[.@k];
	return 0;

// Removes a mission (and its paired quest) from the quest log.
S_Drop:
	if (checkquest(.q_id[getarg(0)], HAVEQUEST) >= 0)
		erasequest .q_id[getarg(0)];
	if (.q_id2[getarg(0)] && checkquest(.q_id2[getarg(0)], HAVEQUEST) >= 0)
		erasequest .q_id2[getarg(0)];
	if (.q_need$[getarg(0)] != "")
		setd .q_need$[getarg(0)], 0;
	return;

// board, quest, cooldown, base level, area, title, goal, base EXP,
// job EXP, Eden Coins, "item,count,..." handed in, "item,count,..." given.
// .q_id2 is a second quest given and paid with the row's quest, for a hunt
// with more targets than one quest can hold. .q_give$ lists items handed out
// on accept ("item,count,..."), and .q_need$/.q_needn a character variable
// that must reach a value before the mission counts as finished.
S_Row:
	.@n = .rows;
	.q_board[.@n] = getarg(0);
	.q_id[.@n] = getarg(1);
	.q_cd[.@n] = getarg(2);
	.q_lv[.@n] = getarg(3);
	.q_area$[.@n] = getarg(4);
	.q_title$[.@n] = getarg(5);
	.q_goal$[.@n] = getarg(6);
	.q_exp[.@n] = getarg(7);
	.q_jexp[.@n] = getarg(8);
	.q_coin[.@n] = getarg(9);
	.q_items$[.@n] = getarg(10);
	.q_rewards$[.@n] = getarg(11);
	if (!.first[getarg(0)])
		.first[getarg(0)] = .@n;
	.last[getarg(0)] = .@n + 1;
	.rows = .@n + 1;
	return;

// Shortens a mission's name in the board menu, which wraps (and overlaps
// the next entry) past about 40 characters. The mission page still shows
// the full quest log title. getarg(0) = quest id, 1 = short name.
// The short names are the mod's own, not kRO text.
S_Menu:
	.@n = inarray(.q_id, getarg(0));
	if (.@n < 1)
		debugmes "Eden Mission Board: no row for quest " + getarg(0);
	else
		.q_menu$[.@n] = getarg(1);
	return;

// Cave shortcuts: warp members who hold one of the listed board hunts.
// getarg(0) = NPC name, 1-3 = map, x, y, 4 = place name, 5.. = quest ids.
function	script	F_EdenMB_Shortcut	{
	mes "[" + getarg(0) + "]";
	for (.@i = 5; .@i < getargcount(); ++.@i) {
		if (checkquest(getarg(.@i), HAVEQUEST) == 1)
			break;
	}
	if (.@i == getargcount()) {
		mes "I take Eden Group members on board missions straight into " + getarg(4) + ".";
		mes "You don't have one of those missions right now.";
		close;
	}
	mes "You're on an Eden Group mission? I can take you straight into " + getarg(4) + ".";
	next;
	if (select("Take me there:Not now") == 2)
		close;
	warp getarg(1), getarg(2), getarg(3);
	end;
}
