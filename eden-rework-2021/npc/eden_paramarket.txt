// A shop NPC: greeting, then the shop. getarg(0) = name, getarg(1) = line,
// getarg(2) = shop.
function	script	F_EdenRwPmShop	{
	mes "[" + getarg(0) + "]";
	mes getarg(1);
	close2;
	callshop getarg(2), 1;
	end;
}

// An exchange NPC: pick what to pay with, then that currency's barter shop.
// getarg(0) = name, getarg(1) = shop name prefix, getarg(2..) = currency items.
function	script	F_EdenRwPmExchange	{
	.@n = getargcount() - 2;
	for (.@i = 0; .@i < .@n; .@i++)
		.@menu$ += getitemname(getarg(.@i + 2)) + ":";
	mes "[" + getarg(0) + "]";
	mes "What will you trade in?";
	next;
	.@s = select(.@menu$ + "Cancel") - 1;
	if (.@s == .@n)
		close;
	close2;
	callshop getarg(1) + getarg(.@s + 2);
	end;
}
