//===== Eden Group 2021 rework: stand down for World Sync Extras =====
//= World Sync Extras (Lil Art) has its own take on parts of the Eden
//= Group: its Gramps stands on the Logistics Officer's cell
//= (moc_para01 29,35), its town quests share the mission boards' quest
//= ids, and it has its own crates and ghost fires for two of the board
//= missions. With both mods on, NPCs stack on one cell and players get
//= two half-linked versions of the same content.
//=
//= So while World Sync Extras is on, this mod stands down completely
//= (Steven, 2026-11-03): every NPC and warp it adds is disabled, the
//= stock Eden NPCs it would hide stay visible, and its premium drops
//= stop (a disabled NPC's OnNPCKillEvent doesn't run). The server is
//= then stock Eden plus World Sync Extras, as if this mod were off.
//= Its tables (items, quests, monsters, barters) still load, since a
//= script can't unload them; nothing uses them while its NPCs are off.
//============================================================

// Returns 1 while World Sync Extras is switched on. F_ModSetting only
// lists switched-on mods and gives the fallback otherwise, so any one of
// its settings answering something other than -1 means it is on. Several
// keys, so a renamed setting on their side doesn't hide it.
function	script	F_Eden_StandDown	{
	setarray .@key$, "city_investment", "town_quests_2021", "gramps_turn_ins", "iro_systems", "placeholders", "npc_mover";
	for (.@i = 0; .@i < getarraysize(.@key$); .@i++)
		if (callfunc("F_ModSetting", "world-sync-extras", .@key$[.@i], -1) != -1)
			return 1;
	return 0;
}

-	script	#eden_rework_stand_down	-1,{
	end;
OnInit:
	if (!callfunc("F_Eden_StandDown"))
		end;
	setarray .@npc$,
